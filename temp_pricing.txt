              </Card.Section>
              <div 
                style={{ padding: '16px', borderTop: '1px solid #dfe3e8', cursor: 'pointer', display: 'flex', justifyContent: 'space-between', alignItems: 'center' }} 
                onClick={() => setPricingExpanded(!pricingExpanded)}
              >
                <Text variant="bodySm" color="subdued">
                   {[compareAtPrice ? 'Compare-at price' : '', showUnitPrice ? 'Unit price' : '', taxable ? 'Charge tax' : '', costPerItem ? 'Cost per item' : ''].filter(Boolean).join(', ') || 'Compare-at price, Unit price, Charge tax, Cost per item'}
                </Text>
                <div style={{display: 'flex', alignItems: 'center'}}>

                    <path d="M10 14a.997.997 0 0 1-.707-.293l-5-5a.999.999 0 1 1 1.414-1.414L10 11.586l4.293-4.293a.999.999 0 1 1 1.414 1.414l-5 5A.997.997 0 0 1 10 14z"/>
                  </svg>
                </div>
              </div>
              
              {pricingExpanded && (
                <Card.Section>
                  <Text variant="headingSm" as="h6">Additional display prices</Text>
                  <div style={{ marginTop: '16px' }}>
                    <TextField label="Compare at price" type="number" value={compareAtPrice} onChange={setCompareAtPrice} prefix={currency} autoComplete="off" />
                  </div>
                  
                  <div style={{ marginTop: '16px' }}>
                    <div>
                      <Popover
                        active={unitPriceActive}
                        activator={
                          <div style={{display: 'flex', alignItems: 'center', gap: '8px', cursor: 'pointer'}} onClick={() => setUnitPriceActive(true)}>
                            <Checkbox label="Show unit price for this product" checked={showUnitPrice} onChange={(v) => { setShowUnitPrice(v); if(v) setUnitPriceActive(true); }} />
                            {showUnitPrice && <Button plain onClick={() => setUnitPriceActive(true)}>Edit</Button>}
                          </div>
                        }
                        onClose={() => setUnitPriceActive(false)}
                        autofocusTarget="none"
                        preventFocusOnClose
                      >

                          <FormLayout>
                            <FormLayout.Group>
                              <TextField label="Total amount" type="number" value={unitTotalAmount} onChange={setUnitTotalAmount} autoComplete="off" />
                              <TextField label="Base measure" type="number" value={unitBaseMeasure} onChange={setUnitBaseMeasure} autoComplete="off" />
                              <Select 
                                label="Unit" 
                                options={[
                                  {label: 'g', value: 'G'},
                                  {label: 'kg', value: 'KG'},
                                  {label: 'ml', value: 'ML'},
                                  {label: 'L', value: 'L'},
                                  {label: 'm', value: 'M'},
                                  {label: 'cm', value: 'CM'}
                                ]} 
                                value={unitBaseUnit} 
                                onChange={setUnitBaseUnit} 
                              />
                            </FormLayout.Group>
                          </FormLayout>
                          <div style={{ marginTop: '16px', display: 'flex', justifyContent: 'flex-end', gap: '8px' }}>
                            <Button onClick={() => { setShowUnitPrice(false); setUnitPriceActive(false); }}>Clear</Button>
                            <Button onClick={() => setUnitPriceActive(false)}>Cancel</Button>
                            <Button primary onClick={() => setUnitPriceActive(false)}>Done</Button>
                          </div>
                        </div>
                      </Popover>
                    </div>
                  </div>
                  
                  <div style={{ marginTop: '16px', paddingTop: '16px', borderTop: '1px solid #dfe3e8' }}>
                    <Checkbox label="Charge tax on this product" checked={taxable} onChange={setTaxable} />
                  </div>

                  <div style={{ marginTop: '16px', paddingTop: '16px', borderTop: '1px solid #dfe3e8' }}>
                    <FormLayout.Group>
                      <TextField label="Cost per item" type="number" value={costPerItem} onChange={setCostPerItem} prefix={currency} autoComplete="off" helpText="Customers won't see this" />
                      <div>
                        <div style={{ marginBottom: '4px', color: '#5c5f62' }}>Profit</div>
                        <div style={{ fontSize: '16px' }}>{price && costPerItem ? (parseFloat(price) - parseFloat(costPerItem)).toFixed(2) : '--'}</div>
                      </div>
                      <div>
                        <div style={{ marginBottom: '4px', color: '#5c5f62' }}>Margin</div>
                        <div style={{ fontSize: '16px' }}>{price && costPerItem && parseFloat(price) > 0 ? ((parseFloat(price) - parseFloat(costPerItem)) / parseFloat(price) * 100).toFixed(1) + '%' : '--'}</div>
                      </div>
                    </FormLayout.Group>
                  </div>
                </Card.Section>
              )}
